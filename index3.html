import osmnx as ox import networkx as nx # ۱. دریافت گراف شبکه جاده‌ای واقعی (برای خودروهای سنگین/معمولی) # این بخش دیتا را مستقیماً از OpenStreetMap دانلود و تبدیل به گراف ریاضی می‌کند place_name = "Oulu, Finland" G = ox.graph_from_place(place_name, network_type='drive') # ۲. تعریف موقعیت مگاپروژه (تقاضا) و شرکت‌های محلی (عرضه) mega_project_coords = (65.0121, 25.4651) # مختصات فرضی سایت هیدروژن target_demand = 12000 # ظرفیت مورد نیاز EPC smes = [ {"name": "Oulu Fab Oy", "coords": (65.0500, 25.4000), "capacity": 5000}, {"name": "Kempele Industrial", "coords": (64.9120, 25.5030), "capacity": 4000}, {"name": "Haukipudas Assembly", "coords": (65.1760, 25.3520), "capacity": 6000} ] # ۳. انطباق مختصات جغرافیایی روی نزدیک‌ترین گره‌های (Nodes) شبکه جاده‌ای demand_node = ox.distance.nearest_nodes(G, X=mega_project_coords[1], Y=mega_project_coords[0]) # ۴. محاسبه مسافت/زمان واقعی روی جاده برای هر SME for sme in smes: sme_node = ox.distance.nearest_nodes(G, X=sme["coords"][1], Y=sme["coords"][0]) # محاسبه کوتاه‌ترین مسیر بر اساس طول واقعی جاده (weight='length') # می‌توان وزن را به 'travel_time' تغییر داد تا زمان رانندگی محاسبه شود network_distance = nx.shortest_path_length(G, sme_node, demand_node, weight='length') sme['road_distance_meters'] = network_distance # ۵. مرتب‌سازی بر اساس نزدیک‌ترین مسافت واقعی جاده‌ای (نه خط صاف) smes.sort(key=lambda x: x['road_distance_meters']) # ۶. تخصیص ظرفیت (Location-Allocation) allocated_capacity = 0 selected_smes = [] for sme in smes: if allocated_capacity < target_demand: selected_smes.append(sme) allocated_capacity += sme['capacity'] print("--- Selected SMEs based on REAL Road Network ---") for s in selected_smes: print(f"{s['name']}: {s['road_distance_meters']/1000:.1f} km via road, Capacity: {s['capacity']}")
